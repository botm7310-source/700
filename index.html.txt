<script>
const API_URL = "/api/chat";
const chat = document.getElementById("chat");
const input = document.getElementById("prompt");
const send = document.getElementById("send");
let lastAnswer = "";

function addMessage(text, type) {
 const welcome = document.getElementById("welcome");
 if(welcome) welcome.remove();

 const row = document.createElement("div");
 row.className = "message " + type;

 const avatar = document.createElement("div");
 avatar.className = "avatar";
 avatar.textContent = type === "user" ? "👤" : "✧";

 const bubble = document.createElement("div");
 bubble.className = "bubble";
 bubble.textContent = text;

 row.append(avatar,bubble);
 chat.appendChild(row);
 chat.scrollTop = chat.scrollHeight;
 return bubble;
}

async function sendMessage() {
 const message = input.value.trim();
 if(!message || send.disabled) return;

 addMessage(message,"user");
 input.value = "";
 send.disabled = true;

 const answerBox = addMessage("جاري التفكير...","bot");

 try {
  const response = await fetch(API_URL,{
   method:"POST",
   headers:{"Content-Type":"application/json"},
   body:JSON.stringify({message})
  });

  const data = await response.json();
  if(!response.ok) throw new Error(data.error || "فشل الاتصال");

  if(!data.reply) throw new Error("لم تصل إجابة");
  answerBox.textContent = data.reply;
  lastAnswer = data.reply;

 } catch(error) {
  answerBox.textContent = "حدث خطأ: " + error.message;
 } finally {
  send.disabled = false;
  chat.scrollTop = chat.scrollHeight;
 }
}

send.addEventListener("click",sendMessage);
input.addEventListener("keydown",e=>{
 if(e.key === "Enter" && !e.shiftKey){
  e.preventDefault();
  sendMessage();
 }
});

document.getElementById("clear").onclick = () => {
 chat.innerHTML = '<div class="welcome" id="welcome"><strong>محادثة جديدة</strong>كيف يمكنني مساعدتك؟</div>';
 lastAnswer = "";
};

document.getElementById("copy").onclick = async () => {
 if(!lastAnswer) return alert("لا توجد إجابة لنسخها");
 try {
  await navigator.clipboard.writeText(lastAnswer);
  alert("تم نسخ الإجابة");
 } catch {
  alert("تعذر النسخ");
 }
};
</script>
